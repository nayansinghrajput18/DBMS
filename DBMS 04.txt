mysql> SELECT subject, AVG(marks) AS Avg_Marks
    -> FROM student
    -> GROUP BY subject;

mysql> SELECT dept,
    -> MAX(Marks) AS Highest_Marks,
    -> MIN(Marks) AS Lowest_Marks
    -> FROM student
    -> GROUP BY dept;

mysql> SELECT dept , sum(marks) as sum_marks from student group by dept;

mysql> SELECT dept, AVG(marks) AS Avg_Marks
    -> FROM student
    -> GROUP BY dept
    -> HAVING AVG(marks) > 80;
